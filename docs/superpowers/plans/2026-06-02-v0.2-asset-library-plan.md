```ts
export function update(object: THREE.Object3D, node: SceneNode): void {
  const data = requireMeshData(node);
  if (!(object instanceof THREE.Mesh)) return;
  object.userData.assetId = data.asset_id;
  const want = data.geometry.kind;
  if (object.userData.geometryKind !== want) {
    object.geometry.dispose();
    object.geometry = geometryFor(want);
    object.userData.geometryKind = want;
  }
  if (object.material instanceof THREE.MeshStandardMaterial) {
    applyOverrides(object.material, data.material_overrides?.[0]);
  }
}
```

```ts
import type { SceneNode } from "@/core/scene/types";
import { generateUUID } from "../../id/uuid";
import type { Command, SceneEditorStore } from "../types";

export const ADD_NODE = "node.add" as const;

export interface AddNodeInput {
  node: SceneNode;
  id?: string;
  timestamp?: number;
}

export class AddNodeCommand implements Command {
  readonly id: string;
  readonly type = ADD_NODE;
  readonly timestamp: number;
  readonly payload: { node: SceneNode } & Record<string, unknown>;
  constructor(input: AddNodeInput) {
    this.id = input.id ?? generateUUID();
    this.timestamp = input.timestamp ?? Date.now();
    this.payload = { node: input.node };
  }
  apply(store: SceneEditorStore): void {
    store.addNode(this.payload.node);
  }
  revert(store: SceneEditorStore): void {
    store.removeNodeSubtree(this.payload.node.id);
  }
  canMergeWith(): boolean {
    return false;
  }
  mergeWith(): AddNodeCommand {
    throw new Error("AddNodeCommand: never merges");
  }
}
```

```ts
export interface LibraryItem {
  id: string;
  category: "geometry" | "light" | "upload";
  nameKey?: string; // i18n key under editor:library.item.*
  name?: string; // literal (uploads use filename)
  icon: LucideIcon;
  makeNode: () => SceneNode;
}

const IDENTITY = {
  position: [0, 0, 0] as [number, number, number],
  rotation: [0, 0, 0, 1] as [number, number, number, number],
  scale: [1, 1, 1] as [number, number, number],
};

function baseNode(name: string, data: SceneNode["data"]): SceneNode {
  return {
    id: generateUUID(),
    name,
    type: data.type as SceneNode["type"],
    transform: IDENTITY,
    parent_id: null,
    children_ids: [],
    visible: true,
    locked: false,
    data,
    behaviors: [],
    user_data: {},
  };
}

const geo = (
  kind: "box" | "sphere" | "plane" | "cylinder",
  name: string,
  icon: LucideIcon,
  id: string,

  id,
  category: "geometry",
  nameKey: `library.item.${kind}`,
  icon,
  makeNode: () => baseNode(name, { type: "mesh", geometry: { kind } }),
});

const light = (
  light_kind: "directional" | "point" | "spot" | "ambient",
  name: string,
  icon: LucideIcon,
  id: string,

): LibraryItem => ({
  id,
  category: "light",
  nameKey: `library.item.${light_kind}`,
  icon,
  makeNode: () =>
    baseNode(name, {
      type: "light",
      light_kind,
      color: "#ffffff",
      intensity: 1,
      ...extra,
    }),
});

export const BUILTIN_LIBRARY_ITEMS: LibraryItem[] = [

}
```
